# C2B Console v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add voice-note input (in-browser Whisper + hosted fast mode), a public send-to-pipeline inbox, and a Linux/macOS-skinned redesign to the hosted C2B console.

**Architecture:** One static page (`Portfolio/c2b/index.html`, self-contained) + one Cloudflare Worker with an R2 queue (`Portfolio/workers/c2b-transcribe/`) + one queue-drainer command in Chat2Business (`crew/pull.py`) that feeds the existing watched-folder pipeline. Spec: `Portfolio/docs/superpowers/specs/2026-08-24-c2b-console-v2-design.md`.

**Tech Stack:** Vanilla JS single-file page; `@huggingface/transformers@4.2.0` via jsDelivr (WebGPU/WASM); Cloudflare Workers + Workers AI (`@cf/openai/whisper-large-v3-turbo`) + R2 + KV; Python 3 stdlib (`urllib`) for `crew/pull.py`; pytest.

## Global Constraints

- The page must remain ONE self-contained HTML file; the only external script source allowed is `cdn.jsdelivr.net` (transformers import inside the inline Web Worker).
- No visitor data is stored server-side except inbox submissions (stored in R2 until pulled). Privacy footer must state all three data paths exactly (local = never leaves device; fast mode = transits Worker, not stored; inbox = stored until pulled).
- Inbox guards (verbatim from spec): audio file ≤ 10 MB, content-type `audio/*` (or `video/mp4|webm` for phone containers), note ≤ 2 KB, ~5 inbox posts/day/IP, queue cap 200 objects.
- OS skin: same `localStorage` key **`ckOS`** as the main site, values `linux` (default) and `macos`; `body.macos` class convention.
- Worker CORS allowlist: `https://captkernel.com`, `https://www.captkernel.com`, `http://localhost:8000`, `http://127.0.0.1:8000`.
- Secrets: `PULL_SECRET` lives as a Worker secret and locally in `C2B_PULL_SECRET` env / `Chat2Business/.secrets/pull.env` (gitignored). Never committed.
- Commits: Worker + page work commits to `Portfolio` (branch `c2b-v2`); crew work commits to `Chat2Business` (branch `feat/console-inbox-pull`). Do not push without asking.
- Executors of Tasks 1–2 should load the `wrangler` skill first; executors of Task 6 must load the `taste-skill` skill first.

```js
/* c2b-transcribe: fast-mode transcription + public inbox for the C2B console. */

const ALLOWED_ORIGINS = new Set([
  "https://captkernel.com",
  "https://www.captkernel.com",
  "http://localhost:8000",
  "http://127.0.0.1:8000",
]);
const MAX_AUDIO_BYTES = 10 * 1024 * 1024;       // inbox cap (spec)
const MAX_CHUNK_BYTES = 2 * 1024 * 1024;        // /transcribe WAV chunk cap
const MAX_NOTE_BYTES = 2048;
const QUEUE_CAP = 200;
const INBOX_PER_DAY = 5;
const TRANSCRIBE_PER_DAY = 120;                  // ~1h of audio in 30s chunks
const AUDIO_TYPES = /^(audio\/|video\/(mp4|webm))/;

function cors(origin) {
  const o = ALLOWED_ORIGINS.has(origin) ? origin : "https://captkernel.com";
  return {
    "Access-Control-Allow-Origin": o,
    "Access-Control-Allow-Methods": "GET,POST,DELETE,OPTIONS",
    "Access-Control-Allow-Headers": "Authorization,Content-Type",
    "Access-Control-Max-Age": "86400",
  };
}
const json = (obj, status, hdrs) =>
  new Response(JSON.stringify(obj), { status: status || 200,
    headers: { "content-type": "application/json", ...hdrs } });

async function ipHash(req, env) {
  const ip = req.headers.get("cf-connecting-ip") || "0.0.0.0";
  const buf = await crypto.subtle.digest("SHA-256",
    new TextEncoder().encode(ip + (env.PULL_SECRET || "salt")));
  return [...new Uint8Array(buf)].slice(0, 8)
    .map(b => b.toString(16).padStart(2, "0")).join("");
}

/* KV daily counter. Returns true when the caller is over the limit. */
async function rateLimited(env, route, hash, limit) {
  const day = new Date().toISOString().slice(0, 10);
  const key = `rl:${route}:${hash}:${day}`;
  const n = parseInt((await env.RATE.get(key)) || "0", 10);
  if (n >= limit) return true;
  await env.RATE.put(key, String(n + 1), { expirationTtl: 86400 });
  return false;
}

function authed(req, env) {
  return req.headers.get("Authorization") === `Bearer ${env.PULL_SECRET}`;
}

function b64(buf) {
  const bytes = new Uint8Array(buf);
  let s = "";
  for (let i = 0; i < bytes.length; i += 0x8000)
    s += String.fromCharCode.apply(null, bytes.subarray(i, i + 0x8000));
  return btoa(s);
}

const sanitize = name =>
  (name || "note").replace(/[^a-zA-Z0-9._-]+/g, "_").slice(0, 80);

export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    const hdrs = cors(req.headers.get("Origin") || "");
    if (req.method === "OPTIONS") return new Response(null, { status: 204, headers: hdrs });

    try {
      if (url.pathname === "/transcribe" && req.method === "POST")
        return await transcribe(req, env, hdrs);
      if (url.pathname === "/inbox" && req.method === "POST")
        return await inboxPost(req, env, hdrs);
      if (url.pathname === "/inbox" && req.method === "GET")
        return authed(req, env) ? await inboxList(env, hdrs)
                                : json({ error: "unauthorized" }, 401, hdrs);
      const m = url.pathname.match(/^\/inbox\/([A-Za-z0-9-]+)$/);
      if (m && (req.method === "GET" || req.method === "DELETE")) {
        if (!authed(req, env)) return json({ error: "unauthorized" }, 401, hdrs);
        return req.method === "GET" ? await inboxGet(env, m[1], hdrs)
                                    : await inboxDelete(env, m[1], hdrs);
      }
      return json({ error: "not found" }, 404, hdrs);
    } catch (e) {
      return json({ error: String(e && e.message || e) }, 500, hdrs);
    }
  },
};

async function transcribe(req, env, hdrs) {
  const hash = await ipHash(req, env);
  if (await rateLimited(env, "t", hash, TRANSCRIBE_PER_DAY))
    return json({ error: "daily transcription limit reached — use local mode" }, 429, hdrs);
  const buf = await req.arrayBuffer();
  if (buf.byteLength === 0 || buf.byteLength > MAX_CHUNK_BYTES)
    return json({ error: "chunk must be 1B–2MB of WAV audio" }, 413, hdrs);
  const out = await env.AI.run("@cf/openai/whisper-large-v3-turbo", { audio: b64(buf) });
  return json({ text: (out && out.text || "").trim() }, 200, hdrs);
}

async function inboxPost(req, env, hdrs) {
  const hash = await ipHash(req, env);
  if (await rateLimited(env, "i", hash, INBOX_PER_DAY))
    return json({ error: "daily submission limit reached (5/day)" }, 429, hdrs);

  const listed = await env.INBOX.list({ prefix: "inbox/", limit: QUEUE_CAP + 1 });
  // each item = audio object + .json sidecar → cap at 2×QUEUE_CAP keys
  if (listed.objects.length >= QUEUE_CAP * 2)
    return json({ error: "inbox is full — try again in a few days" }, 503, hdrs);

  const form = await req.formData();
  const audio = form.get("audio");
  const note = String(form.get("note") || "").slice(0, MAX_NOTE_BYTES);
  if (!(audio instanceof File)) return json({ error: "missing audio file" }, 400, hdrs);
  if (audio.size > MAX_AUDIO_BYTES) return json({ error: "audio over 10MB" }, 413, hdrs);
  if (!AUDIO_TYPES.test(audio.type || ""))
    return json({ error: "not an audio file" }, 415, hdrs);

  const id = `${Date.now()}-${crypto.randomUUID().slice(0, 8)}`;
  const ext = (audio.name.match(/\.[A-Za-z0-9]{1,5}$/) || [".m4a"])[0].toLowerCase();
  await env.INBOX.put(`inbox/${id}${ext}`, audio.stream(),
    { httpMetadata: { contentType: audio.type } });
  await env.INBOX.put(`inbox/${id}.json`, JSON.stringify({
    id, filename: sanitize(audio.name), note, ip: hash,
    received: new Date().toISOString(), size: audio.size,
  }));
  return json({ id }, 200, hdrs);
}

async function inboxList(env, hdrs) {
  const listed = await env.INBOX.list({ prefix: "inbox/", limit: 1000 });
  const items = [];
  for (const o of listed.objects) {
    if (!o.key.endsWith(".json")) continue;
    const meta = await env.INBOX.get(o.key);
    if (meta) items.push(await meta.json());
  }
  return json({ items }, 200, hdrs);
}

async function findAudioKey(env, id) {
  const listed = await env.INBOX.list({ prefix: `inbox/${id}`, limit: 10 });
  return (listed.objects.find(o => !o.key.endsWith(".json")) || {}).key;
}

async function inboxGet(env, id, hdrs) {
  const key = await findAudioKey(env, id);
  if (!key) return json({ error: "not found" }, 404, hdrs);
  const obj = await env.INBOX.get(key);
  const metaObj = await env.INBOX.get(`inbox/${id}.json`);
  const meta = metaObj ? await metaObj.json() : {};
  return new Response(obj.body, { headers: {
    "content-type": obj.httpMetadata && obj.httpMetadata.contentType || "application/octet-stream",
    "x-c2b-filename": key.slice("inbox/".length),
    "x-c2b-note": encodeURIComponent(meta.note || ""),
    ...hdrs } });
}

async function inboxDelete(env, id, hdrs) {
  const key = await findAudioKey(env, id);
  if (key) await env.INBOX.delete(key);
  await env.INBOX.delete(`inbox/${id}.json`);
  return json({ deleted: true }, 200, hdrs);
}
```

```python
"""Drain the hosted console's R2 inbox into the watched Inbox folder.

The captkernel.com/c2b/ page lets anyone queue a voice note ("send to the C2B
pipeline"). This module downloads each queued note into <base>/Inbox/ — where
crew.watch already ingests audio — then deletes it from the queue. Text notes
land in <base>/Notes/ as sidecars.

Config (either source):
    env  C2B_PULL_URL / C2B_PULL_SECRET
    file .secrets/pull.env  (repo root; gitignored)  KEY=VALUE lines

Usage:
    py -m crew.pull            # drain once
    py -m crew.pull --keep     # download but leave the queue intact (testing)
"""
from __future__ import annotations

import argparse
import json
import re
import sys
import urllib.request
from pathlib import Path

SECRETS_FILE = Path(__file__).resolve().parent.parent / ".secrets" / "pull.env"


def load_config() -> tuple[str, str] | None:
    import os
    url, secret = os.environ.get("C2B_PULL_URL"), os.environ.get("C2B_PULL_SECRET")
    if not (url and secret) and SECRETS_FILE.exists():
        kv = dict(
            line.split("=", 1)
            for line in SECRETS_FILE.read_text(encoding="utf-8").splitlines()
            if "=" in line and not line.lstrip().startswith("#")
        )
        url = url or kv.get("C2B_PULL_URL", "").strip()
        secret = secret or kv.get("C2B_PULL_SECRET", "").strip()
    return (url.rstrip("/"), secret) if url and secret else None


def _default_http(url: str, method: str = "GET", secret: str = ""):
    req = urllib.request.Request(url, method=method,
                                 headers={"Authorization": f"Bearer {secret}"})
    with urllib.request.urlopen(req, timeout=60) as res:  # noqa: S310 — https URL from config
        return res.status, dict(res.headers), res.read()


def _safe_ext(filename: str) -> str:
    m = re.search(r"\.([A-Za-z0-9]{1,5})$", filename or "")
    return f".{m.group(1).lower()}" if m else ".m4a"


def pull_all(base: Path, http=None, keep: bool = False) -> list[Path]:
    """Download every queued note into base/Inbox; return new files (skips known)."""
    cfg = load_config()
    if cfg is None:
        return []
    url, secret = cfg
    http = http or _default_http

    status, _, body = http(f"{url}/inbox", "GET", secret)
    if status != 200:
        raise RuntimeError(f"inbox list failed: HTTP {status}")
    items = json.loads(body).get("items", [])

    inbox, notes = base / "Inbox", base / "Notes"
    inbox.mkdir(parents=True, exist_ok=True)
    got: list[Path] = []
    for it in items:
        iid = it["id"]
        dest = inbox / f"c2b-{iid}{_safe_ext(it.get('filename', ''))}"
        if not dest.exists():
            status, headers, audio = http(f"{url}/inbox/{iid}", "GET", secret)
            if status != 200:
                print(f"[pull] skip {iid}: HTTP {status}", file=sys.stderr)
                continue
            tmp = dest.with_suffix(dest.suffix + ".part")
            tmp.write_bytes(audio)
            tmp.rename(dest)                      # atomic: watcher never sees partials
            note = headers.get("x-c2b-note") or it.get("note") or ""
            note = urllib.parse.unquote(note) if "%" in note else note
            if note.strip():
                notes.mkdir(parents=True, exist_ok=True)
                (notes / f"c2b-{iid}.txt").write_text(note, encoding="utf-8")
            got.append(dest)
        if not keep:
            http(f"{url}/inbox/{iid}", "DELETE", secret)
    return got


def drain_quietly(base: Path) -> None:
    """Best-effort drain for the watcher loop; never raises."""
    try:
        for p in pull_all(base):
            print(f"[pull] fetched {p.name}")
    except Exception as e:  # noqa: BLE001 — network failure must not kill the watcher
        print(f"[pull] WARN: {type(e).__name__}: {e}", file=sys.stderr)


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(prog="crew.pull",
                                 description="Drain the hosted C2B inbox queue.")
    ap.add_argument("--folder", default=None, help="watch base folder override")
    ap.add_argument("--keep", action="store_true", help="don't delete from the queue")
    a = ap.parse_args(argv)
    from .watch import default_base
    base = Path(a.folder) if a.folder else default_base()
    if load_config() is None:
        print("[pull] not configured — set C2B_PULL_URL/C2B_PULL_SECRET "
              f"or create {SECRETS_FILE}", file=sys.stderr)
        return 2
    got = pull_all(base, keep=a.keep)
    print(f"[pull] {len(got)} new note(s) -> {base / 'Inbox'}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```js
async function decodeToPCM(blob){
  const ctx = new (window.AudioContext||window.webkitAudioContext)({sampleRate:16000});
  try{
    const buf = await ctx.decodeAudioData(await blob.arrayBuffer());
    /* average channels to mono */
    const ch0 = buf.getChannelData(0);
    if(buf.numberOfChannels === 1) return ch0.slice();
    const out = new Float32Array(buf.length);
    for(let c=0;c<buf.numberOfChannels;c++){
      const d = buf.getChannelData(c);
      for(let i=0;i<d.length;i++) out[i] += d[i]/buf.numberOfChannels;
    }
    return out;
  } finally { ctx.close(); }
}

function encodeWAV(f32, sampleRate){
  sampleRate = sampleRate || 16000;
  const buf = new ArrayBuffer(44 + f32.length*2), v = new DataView(buf);
  const wstr = (o,s)=>{ for(let i=0;i<s.length;i++) v.setUint8(o+i, s.charCodeAt(i)); };
  wstr(0,"RIFF"); v.setUint32(4, 36+f32.length*2, true); wstr(8,"WAVE");
  wstr(12,"fmt "); v.setUint32(16,16,true); v.setUint16(20,1,true); v.setUint16(22,1,true);
  v.setUint32(24,sampleRate,true); v.setUint32(28,sampleRate*2,true);
  v.setUint16(32,2,true); v.setUint16(34,16,true);
  wstr(36,"data"); v.setUint32(40, f32.length*2, true);
  for(let i=0;i<f32.length;i++){
    const s = Math.max(-1, Math.min(1, f32[i]));
    v.setInt16(44+i*2, s<0 ? s*0x8000 : s*0x7FFF, true);
  }
  return new Blob([buf], {type:"audio/wav"});
}

}
```

```js
async function transcribeFast(pcm, onProgress){
  const chunks = sliceChunks(pcm, 30);
  let text = "";
  for(let i=0;i<chunks.length;i++){
    onProgress("fast mode — chunk "+(i+1)+"/"+chunks.length+"…");
    const res = await fetch(WORKER_URL+"/transcribe", {method:"POST", body: encodeWAV(chunks[i])});
    if(!res.ok){
      const j = await res.json().catch(()=>({}));
      throw new Error("fast mode failed (HTTP "+res.status+"): "+(j.error||res.statusText));
    }
    text += (text?" ":"") + ((await res.json()).text||"");
  }
  return text.trim();
}
```
